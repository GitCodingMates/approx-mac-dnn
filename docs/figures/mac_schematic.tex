% Register-transfer schematic of rtl/mac.v (MUL = 1, K = 8), drawn from the
% Yosys cells of Fig. fig:schem. Requires tikz with arrows.meta.
\begin{tikzpicture}[
    font=\footnotesize,
    >={Stealth[length=1.8mm]},
    bus/.style={very thick, ->},
    sig/.style={thin, ->},
    cell/.style={draw, thick, fill=blue!6, align=center, minimum height=1.3cm},
    ff/.style={draw, thick, fill=yellow!22, minimum width=1.25cm,
               minimum height=1.3cm, align=center, font=\scriptsize\ttfamily},
    port/.style={font=\scriptsize\ttfamily, text=black!70},
    w/.style={font=\scriptsize, fill=white, inner sep=1pt},
]
% inputs
\node[port, anchor=east] (a)   at (0,1.3)  {a[7:0]};
\node[port, anchor=east] (b)   at (0,0.5)  {b[7:0]};
\node[port, anchor=east] (en)  at (0,-2.0) {en};
\node[port, anchor=east] (clr) at (0,-3.6) {clr};
\node[port, anchor=east] (rst) at (0,-4.8) {rst};
\node[port, anchor=east] (clk) at (0,-5.3) {clk};

% stage 1
\node[cell, minimum width=2.1cm] (mul) at (1.8,0.9)
     {\texttt{mul\_trunc}\\$K=8$};
\draw[bus] (a) -- (a -| mul.west);
\draw[bus] (b) -- (b -| mul.west);

\node[ff] (pq) at (4.7,0.9) {p\_q\\[2pt]\$sdff};
\draw[bus] (mul.east) -- node[w, above]{$p$\,/16} (pq.west);

\node[ff] (vq)   at (4.7,-2.0) {v\_q\\[2pt]\$sdff};
\node[ff] (clrq) at (4.7,-3.6) {clr\_q\\[2pt]\$sdff};
\draw[sig] (en)  -- (vq.west);
\draw[sig] (clr) -- (clrq.west);

% stage 2: sign extension
\node[cell, minimum width=1.5cm, minimum height=0.9cm, fill=gray!10] (sx)
     at (7.1,0.9) {sign\\extend};
\draw[bus] (pq.east) -- node[w, above]{16} (sx.west);

% clear multiplexer
\coordinate (m) at (7.1,-0.9);
\draw[thick, fill=blue!6] ($(m)+(-0.35,0.65)$) -- ($(m)+(0.35,0.4)$)
     -- ($(m)+(0.35,-0.4)$) -- ($(m)+(-0.35,-0.65)$) -- cycle;
\node[font=\scriptsize] at ($(m)+(-0.18,0.35)$) {0};
\node[font=\scriptsize] at ($(m)+(-0.18,-0.35)$) {1};
\node[font=\scriptsize\ttfamily, above=2pt] at ($(m)+(0,0.6)$) {\$mux};

% adder
\node[draw, thick, circle, fill=blue!6, minimum size=0.8cm] (add) at (9.0,0.2) {$+$};
\draw[bus] (sx.east) -| node[w, pos=0.25, above]{32} (add.north);
\draw[bus] ($(m)+(0.35,0)$) -| (add.south);

% accumulator register
\node[ff, minimum height=1.6cm] (acc) at (11.0,0.2) {acc\\[2pt]\$sdffe\\[2pt]EN};
\draw[bus] (add.east) -- node[w, above]{32} (acc.west);
\node[port, anchor=west] (out) at (13.0,0.2) {acc[31:0]};
\draw[bus] (acc.east) -- (out.west);

% feedback acc -> mux input 0, constant 0 -> input 1
\draw[bus] (12.3,0.2) -- (12.3,-3.95) -- (6.3,-3.95) |- ($(m)+(-0.35,0.35)$);
\draw[bus] (6.6,-1.25) node[left, font=\scriptsize, inner sep=1pt]{0} -- ($(m)+(-0.35,-0.35)$);
\fill (12.3,0.2) circle (1.4pt);

% control
\draw[sig] (clrq.east) -| ($(m)+(0,-0.52)$);
\draw[sig] (vq.east) -| (acc.south);

% clock and synchronous reset (shared by all four registers)
\draw[thin, dashed, black!60] (rst) -- (13.0,-4.8)
      node[right, font=\scriptsize, text=black!60]{SRST of all registers};
\draw[thin, dashed, black!60] (clk) -- (13.0,-5.3)
      node[right, font=\scriptsize, text=black!60]{CLK of all registers};

% stage labels
\node[font=\bfseries\small, text=blue!55!black]  at (2.9,2.3) {Stage 1: multiply};
\node[font=\bfseries\small, text=green!40!black] at (9.6,2.3) {Stage 2: accumulate};
\draw[densely dotted, black!40] (6.1,2.6) -- (6.1,-4.3);
\end{tikzpicture}
